function pctInRange(u,type){
  const st=settingsOf(u),cut=Date.now()-30*864e5;
  const arr=readings(u).filter(r=>r.type===type&&new Date(r.date).getTime()>=cut);
  if(!arr.length)return null;

    ?arr.filter(r=>r.s<=st.sys&&r.d<=st.dia&&r.s>=90&&r.d>=60).length
    :arr.filter(r=>classifyGlu(r.g,r.ctx)[0]==='Normal').length;
  return Math.round(ok/arr.length*100);
}

  const rec=gl.slice(0,5),g=avg(rec,r=>r.g);
  if(g>st.glu+25)return `Promedio reciente de <b>${g} mg/dL</b>, por encima de tu objetivo (${st.glu} mg/dL). Revisa comidas, actividad física y horarios.`;
  return `Promedio reciente de <b>${g} mg/dL</b>. ${g<=st.glu?`Dentro de tu objetivo (${st.glu}). ¡Sigue así! ✨`:`Cerca de tu objetivo (${st.glu}). Vas por buen camino.`}`;
}
function analysisTA(u,ta,st){
  if(ta.length<2)return 'Registra al menos dos mediciones de tensión para generar un análisis.';
  const s=avg(ta,r=>r.s),d=avg(ta,r=>r.d);
  const withP=ta.filter(r=>r.p!=null);const p=avg(withP,r=>r.p);
  const ok=ta.filter(r=>r.s<=st.sys&&r.d<=st.dia).length,pct=Math.round(ok/ta.length*100);
  const crisis=ta.filter(r=>r.s>=180||r.d>=120).length;
  let out=`Con <b>${ta.length}</b> mediciones, tu promedio es <b>${s}/${d} mmHg</b>${p!=null?` (pulso medio ${p} lpm)`:''}. El <b>${pct}%</b> está dentro de tu meta ${st.sys}/${st.dia}.`;
  if(crisis)out+=` <b class="warn">⚠️ ${crisis} lectura${crisis>1?'s':''} en rango de crisis: requiere valoración médica.</b>`;
  else if(pct>=70)out+=' Gran constancia y buen control. 🏆';
  else out+=' Identifica los momentos del día con valores más altos y compártelos con tu médico.';
  return out;
}
function analysisGlu(u,gl,st){
  if(gl.length<2)return 'Registra al menos dos mediciones de glucosa para analizar tu tendencia.';
  const g=avg(gl,r=>r.g);
  const lows=gl.filter(r=>r.g<70).length;
  const highs=gl.filter(r=>r.g>=180||(/ayunas|antes/i.test(r.ctx)&&r.g>=126)).length;
  const ok=gl.filter(r=>classifyGlu(r.g,r.ctx)[0]==='Normal').length;
  const pct=Math.round(ok/gl.length*100);
  let out=`Con <b>${gl.length}</b> mediciones, tu promedio es <b>${g} mg/dL</b> y el <b>${pct}%</b> está en rango normal.`;
  if(lows)out+=` <b class="warn">⚠️ ${lows} lectura${lows>1?'s':''} baja${lows>1?'s':''} (&lt;70): revisa horarios de comida y medicación.</b>`;
  if(highs)out+=` Se registraron ${highs} valor${highs>1?'es':''} alto${highs>1?'s':''}: compártelos con tu equipo de salud.`;
  if(!lows&&!highs)out+=' Excelente control. ✨';
  return out;
}

}

    }
    const morns=ta.filter(r=>r.m==='Mañana'),nights=ta.filter(r=>r.m==='Noche');

    const crisis=ta.filter(r=>r.s>=180||r.d>=120).length;

    pgPct=Math.round(okg/gl.length*100);

    const fasts=gl.filter(r=>/ayunas/i.test(r.ctx)),posts=gl.filter(r=>/despu/i.test(r.ctx));
    const gf=avg(fasts,r=>r.g),gp=avg(posts,r=>r.g);

    html.push(`<div class="rep-sec"><h4>💧 Glucosa · ${gl.length} mediciones</h4>
      <p>Promedio general <b>${g} mg/dL</b>. Hemoglobina glicosilada estimada (eA1c): <b>${a1c.toFixed(1)}%</b> (${a1cTxt}) — cálculo orientativo, no reemplaza un examen de laboratorio.</p>
      <p>Tiempo en rango: <b>${pgPct}%</b> de las lecturas en nivel normal según su contexto.</p>
      <p>Eventos: <b>${lowsArr.length}</b> hipoglucemia(s) (&lt;70 mg/dL) y <b>${highsArr.length}</b> valor(es) muy alto(s) (≥180 mg/dL).</p>

    plain.push('');

    if(lowsArr.length)recs.push(`Tienes ${lowsArr.length} hipoglucemia(s) registrada(s): lleva siempre carbohidratos de absorción rápida y revisa horarios de comida o medicación con tu equipo de salud.`);

  }

function gradOf(color){
  return function(context){
    const ch=context.chart,{ctx,chartArea}=ch;
    if(!chartArea)return color+'22';
    const g=ctx.createLinearGradient(0,chartArea.top,0,chartArea.bottom);
    g.addColorStop(0,color+'44');g.addColorStop(1,color+'00');return g;
  };
}

    $('heroDate').textContent=fmtDT(lt.date);
    const [t,c]=classifyBP(lt.s,lt.d);setPill($('heroStatus'),t,c);
    const dv=pt?lt.s-pt.s:null;
    $('heroDelta').innerHTML=dv==null?'':(dv>0?`<span class="up">▲ +${dv}</span>`:dv<0?`<span class="down">▼ ${dv}</span>`:'<span class="flat">＝ estable</span>')+' <span class="sub">vs. anterior</span>';
    if(lt.p!=null){tween($('pulseVal'),lt.p);const [pt2,pc]=classifyPulse(lt.p);setPill($('pulsePill'),pt2,pc)}
    else{tween($('pulseVal'),null);setPill($('pulsePill'),'—','blue')}
  }else{
    tween($('heroSys'),null);tween($('heroDia'),null);tween($('pulseVal'),null);
    $('heroDate').textContent='sin registros';setPill($('heroStatus'),'Sin datos','blue');setPill($('pulsePill'),'—','blue');
    $('heroDelta').innerHTML='';
  }

  const lg=gl[0];
  if(lg){tween($('gluVal'),lg.g);$('gluDate').textContent=fmtDT(lg.date);const[t,c]=classifyGlu(lg.g,lg.ctx);setPill($('gluPill'),t,c)}
  else{tween($('gluVal'),null);$('gluDate').textContent='sin registros';setPill($('gluPill'),'Sin datos','blue')}

  const w7=r=>Date.now()-new Date(r.date)<7*864e5;
  const tw=ta.filter(w7),gw=gl.filter(w7);
  $('avgTA7').textContent=tw.length?`${avg(tw,r=>r.s)}/${avg(tw,r=>r.d)}`:'—';
  $('avgGlu7').textContent=gw.length?avg(gw,r=>r.g):'—';

  setRing('ringTA','ringTAtxt',pctInRange(u,'bp'));
  setRing('ringGlu','ringGlutxt',pctInRange(u,'glu'));
  $('streakVal').textContent=streakDays(u);
  const now=Date.now(),all=readings(u);
  const wk1=all.filter(r=>now-new Date(r.date)<7*864e5).length;
  const wk0=all.filter(r=>{const dt=now-new Date(r.date);return dt>=7*864e5&&dt<14*864e5}).length;
  $('wkNow').textContent=wk1;$('wkPrev').textContent=wk0;
  $('wkDelta').innerHTML=wk0===0?(wk1>0?'<span style="color:var(--green)">Nuevo impulso ✨</span>':'<span style="color:var(--muted)">Sin actividad</span>'):(wk1>=wk0?`<span style="color:var(--green)">▲ +${Math.round((wk1-wk0)/wk0*100)}%</span>`:`<span style="color:var(--red)">▼ ${Math.round((wk0-wk1)/wk0*100)}%</span>`);

  $('insightTA').innerHTML=insightTA(u,ta,st);
  $('insightGlu').innerHTML=insightGlu(u,gl,st);

  renderHist();

  $('aTA').textContent=ta.length?`${avg(ta,r=>r.s)}/${avg(ta,r=>r.d)}`:'—';
  $('aGlu').textContent=gl.length?avg(gl,r=>r.g):'—';
  const withP=ta.filter(r=>r.p!=null);
  $('aPulse').textContent=withP.length?avg(withP,r=>r.p):'—';
  if(ta.length){const mx=ta.reduce((a,b)=>b.s>a.s?b:a);$('aMaxTA').textContent=`${mx.s}/${mx.d}`}else $('aMaxTA').textContent='—';
  if(gl.length){const mx=gl.reduce((a,b)=>b.g>a.g?b:a);$('aMaxGlu').textContent=mx.g}else $('aMaxGlu').textContent='—';
  $('aCountTA').textContent=ta.length;$('aCountGlu').textContent=gl.length;
  $('aGoals').textContent=`${st.sys}/${st.dia} · ${st.glu}`;
  $('aInsTA').innerHTML=analysisTA(u,ta,st);
  $('aInsGlu').innerHTML=analysisGlu(u,gl,st);

  const [tl7,tp7]=last7prev7(ta);
  if(tl7.length&&tp7.length){
    const s7=avg(tl7,r=>r.s),d7=avg(tl7,r=>r.d),s0=avg(tp7,r=>r.s),d0=avg(tp7,r=>r.d);
    $('trTA').innerHTML=`${s7}/${d7}<span class="arw ${s7<=s0?'good':'bad'}">${s7<=s0?'▼':'▲'}</span><span style="color:var(--muted);font-size:15px">${s0}/${d0}</span>`;
    setPill($('trTApill'),s7<s0-2?'Mejorando':s7>s0+2?'Empeorando':'Estable',s7<s0-2?'green':s7>s0+2?'red':'blue');
  }else{
    $('trTA').textContent=tl7.length?`${avg(tl7,r=>r.s)}/${avg(tl7,r=>r.d)}`:'—';
    setPill($('trTApill'),'Sin comparativa','blue');
  }
  const [gl7,gp7]=last7prev7(gl);
  if(gl7.length&&gp7.length){
    const g7=avg(gl7,r=>r.g),g0=avg(gp7,r=>r.g);
    $('trGlu').innerHTML=`${g7}<span class="arw ${g7<=g0?'good':'bad'}">${g7<=g0?'▼':'▲'}</span><span style="color:var(--muted);font-size:15px">${g0}</span>`;
    setPill($('trGlupill'),g7<g0-4?'Mejorando':g7>g0+4?'Empeorando':'Estable',g7<g0-4?'green':g7>g0+4?'red':'blue');
  }else{
    $('trGlu').textContent=gl7.length?avg(gl7,r=>r.g):'—';
    setPill($('trGlupill'),'Sin comparativa','blue');
  }

  const sdS=stdDev(ta.map(r=>r.s)),sdD=stdDev(ta.map(r=>r.d)),sdG=stdDev(gl.map(r=>r.g));
  $('varS').textContent=sdS!=null?`± ${Math.round(sdS)} mmHg`:'—';
  $('varD').textContent=sdD!=null?`± ${Math.round(sdD)} mmHg`:'—';
  $('varG').textContent=sdG!=null?`± ${Math.round(sdG)} mg/dL`:'—';
  setPill($('varPill'),sdS==null?'Sin datos':sdS<=8?'Variabilidad baja':sdS<=15?'Variabilidad moderada':'Variabilidad alta',sdS==null?'blue':sdS<=8?'green':sdS<=15?'yellow':'red');

  if(gl.length>=3){
    const mg=avg(gl,r=>r.g),a1c=(mg+46.7)/28.7;
    $('ea1c').textContent=a1c.toFixed(1)+'%';
    setPill($('ea1cPill'),a1c<5.7?'Rango normal':a1c<6.5?'Rango elevado':'Muy elevada',a1c<5.7?'green':a1c<6.5?'yellow':'red');
  }else{
    $('ea1c').textContent='—';
    setPill($('ea1cPill'),'Necesitas ≥3 registros','blue');
  }

  $('reportBody').innerHTML=buildReport().html;

  paintAvatar($('pAvatar'),u);
  $('pName').textContent=u.first+' '+u.last;
  $('pMeta').textContent=`${ageFromBirth(u.birth)} años · Nacimiento: ${u.birth||'—'} · ${all.length} medición${all.length!==1?'es':''}`;
  $('pCountTA').textContent=ta.length;$('pCountGlu').textContent=gl.length;
  renderUserList();

  $('tSys').value=st.sys;$('tDia').value=st.dia;$('tGlu').value=st.glu;

  drawChartsFor(currentTab);
}
